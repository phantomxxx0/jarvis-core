+import { Intent } from '../../contracts/intent';
+
+export class IntentNormalizer {
+  static normalize(rawJson: Record<string, unknown>): Intent {
+    return {
+      version: 1,
+      type: typeof rawJson.intent === 'string' ? rawJson.intent : 'UNKNOWN',
+      confidence: Math.max(0, Math.min(1, Number(rawJson.confidence) || 0)), // clamp confidence
+      entities: (rawJson.entities as Record<string, unknown>) || {},
+      goal: typeof rawJson.goal === 'string' ? rawJson.goal : '',
+      constraints: Array.isArray(rawJson.constraints)
+        ? (rawJson.constraints as string[])
+        : [],
+      requiresMemory: !!rawJson.requiresMemory,
+      requiresKnowledge: !!rawJson.requiresKnowledge,
+      requiresPlanning: !!rawJson.requiresPlanning,
+      requiresTools: !!rawJson.requiresTools,
+      capabilities: Array.isArray(rawJson.capabilities)
+        ? (rawJson.capabilities as string[])
+        : ['CHAT'],
+
+      // Backward compatibility during migration:
+      id: '00000000-0000-0000-0000-000000000000',
+      category: typeof rawJson.intent === 'string' ? rawJson.intent : 'UNKNOWN',
+      objective: typeof rawJson.goal === 'string' ? rawJson.goal : '',
+      parameters: {},
+    };
+  }
+
+  static getFallbackIntent(query: string): Intent {
+    return {
+      version: 1,
+      type: 'UNKNOWN',
+      confidence: 1.0,
+      entities: {},
+      goal: query,
+      constraints: [],
+      requiresMemory: false,
+      requiresKnowledge: false,
+      requiresPlanning: true, // fallback to planning if we don't know
+      requiresTools: false,
+      capabilities: ['CHAT'],
+
+      // Backward compatibility during migration:
+      id: '00000000-0000-0000-0000-000000000000',
+      category: 'UNKNOWN',
+      objective: query,
+      parameters: {},
+    };
+  }

+import { Logger } from '@nestjs/common';
+
+export class IntentParser {
+  private static readonly logger = new Logger(IntentParser.name);
+
+  static parse(llmOutput: string): Record<string, unknown> {
+    try {
+      // Sometimes LLMs wrap JSON in markdown code blocks
+      const cleanOutput = llmOutput
+        .replace(/^```json\s*/, '')
+        .replace(/```\s*$/, '')
+        .trim();
+      return JSON.parse(cleanOutput) as Record<string, unknown>;
+    } catch (error) {
+      this.logger.error('Failed to parse LLM output as JSON', error);
+      throw new Error('ParseError');
+    }
+  }

+import { Logger } from '@nestjs/common';
+
+export class IntentValidator {
+  private static readonly logger = new Logger(IntentValidator.name);
+
+  static validate(
+    rawJson: Record<string, unknown> | null | undefined,
+  ): boolean {
+    if (!rawJson || typeof rawJson !== 'object') {
+      this.logger.error('Validation failed: Root is not an object');
+      return false;
+    }
+
+    if (rawJson.version !== 1) {
+      this.logger.error('Validation failed: Unsupported version');
+      return false;
+    }
+
+    if (typeof rawJson.intent !== 'string') {
+      this.logger.error('Validation failed: intent must be a string');
+      return false;
+    }
+
+    const confidence = rawJson.confidence;
+    if (typeof confidence !== 'number' || confidence < 0 || confidence > 1) {
+      this.logger.error(
+        'Validation failed: confidence must be a number between 0 and 1',
+      );
+      return false;
+    }
+
+    if (!rawJson.entities || typeof rawJson.entities !== 'object') {
+      this.logger.error('Validation failed: entities must be an object');
+      return false;
+    }
+
+    if (!Array.isArray(rawJson.capabilities)) {
+      this.logger.error('Validation failed: capabilities must be an array');
+      return false;
+    }
+
+    return true;
+  }
+}
